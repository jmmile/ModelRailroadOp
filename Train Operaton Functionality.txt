from sqlalchemy import (
    select,
)

from modelrailroadops.database.database import SessionLocal

from modelrailroadops.models.operations_session_train import (
    OperationsSessionTrain,
)


class OperationsSessionTrainService:
    """
    Service for creating, retrieving, and deleting
    OperationsSessionTrain records.

    An OperationsSessionTrain record assigns a Train
    to an Operations Session.
    """

    #
    # Get all train assignments
    #

    @staticmethod
    def get_all():

        with SessionLocal() as session:

            statement = (
                select(
                    OperationsSessionTrain
                )
                .order_by(
                    OperationsSessionTrain.id,
                )
            )

            return (
                session.execute(
                    statement
                )
                .scalars()
                .all()
            )

    #
    # Get assignment by ID
    #

    @staticmethod
    def get_by_id(
        assignment_id,
    ):

        if assignment_id is None:

            return None

        with SessionLocal() as session:

            return session.get(
                OperationsSessionTrain,
                assignment_id,
            )

    #
    # Get all trains assigned to an
    # Operations Session.
    #

    @staticmethod
    def get_by_operations_session(
        operations_session_id,
    ):

        if operations_session_id is None:

            return []

        with SessionLocal() as session:

            statement = (
                select(
                    OperationsSessionTrain
                )
                .where(
                    OperationsSessionTrain.operations_session_id
                    == operations_session_id
                )
                .order_by(
                    OperationsSessionTrain.id,
                )
            )

            return (
                session.execute(
                    statement
                )
                .scalars()
                .all()
            )

    #
    # Get all Operations Sessions assigned
    # to a Train.
    #

    @staticmethod
    def get_by_train(
        train_id,
    ):

        if train_id is None:

            return []

        with SessionLocal() as session:

            statement = (
                select(
                    OperationsSessionTrain
                )
                .where(
                    OperationsSessionTrain.train_id
                    == train_id
                )
                .order_by(
                    OperationsSessionTrain.id,
                )
            )

            return (
                session.execute(
                    statement
                )
                .scalars()
                .all()
            )

    #
    # Check whether a Train is already
    # assigned to an Operations Session.
    #

    @staticmethod
    def exists(
        operations_session_id,
        train_id,
    ):

        if (
            operations_session_id is None
            or train_id is None
        ):

            return False

        with SessionLocal() as session:

            statement = (
                select(
                    OperationsSessionTrain
                )
                .where(
                    OperationsSessionTrain.operations_session_id
                    == operations_session_id,
                    OperationsSessionTrain.train_id
                    == train_id,
                )
            )

            return (
                session.execute(
                    statement
                )
                .scalars()
                .first()
                is not None
            )

    #
    # Assign a Train to an Operations Session.
    #

    @staticmethod
    def create(
        operations_session_id,
        train_id,
    ):

        if operations_session_id is None:

            return (
                False,
                "No Operations Session was specified.",
            )

        if train_id is None:

            return (
                False,
                "No train was specified.",
            )

        with SessionLocal() as session:

            #
            # Prevent duplicate assignments.
            #

            existing = (
                session.execute(
                    select(
                        OperationsSessionTrain
                    )
                    .where(
                        OperationsSessionTrain.operations_session_id
                        == operations_session_id,
                        OperationsSessionTrain.train_id
                        == train_id,
                    )
                )
                .scalars()
                .first()
            )

            if existing is not None:

                return (
                    False,
                    (
                        "This train is already "
                        "assigned to the Operations Session."
                    ),
                )

            assignment = OperationsSessionTrain(
                operations_session_id=(
                    operations_session_id
                ),
                train_id=train_id,
            )

            session.add(
                assignment
            )

            session.commit()

            session.refresh(
                assignment
            )

            return (
                True,
                assignment,
            )

    #
    # Delete a train assignment.
    #

    @staticmethod
    def delete(
        assignment_id,
    ):

        if assignment_id is None:

            return (
                False,
                "No train assignment was specified.",
            )

        with SessionLocal() as session:

            assignment = session.get(
                OperationsSessionTrain,
                assignment_id,
            )

            if assignment is None:

                return (
                    False,
                    (
                        f"Operations Session train "
                        f"assignment {assignment_id} "
                        "was not found."
                    ),
                )

            session.delete(
                assignment
            )

            session.commit()

            return (
                True,
                "Train assignment deleted successfully.",
            )

    #
    # Remove a specific Train from an
    # Operations Session.
    #

    @staticmethod
    def delete_by_operations_session_and_train(
        operations_session_id,
        train_id,
    ):

        if operations_session_id is None:

            return (
                False,
                "No Operations Session was specified.",
            )

        if train_id is None:

            return (
                False,
                "No train was specified.",
            )

        with SessionLocal() as session:

            statement = (
                select(
                    OperationsSessionTrain
                )
                .where(
                    OperationsSessionTrain.operations_session_id
                    == operations_session_id,
                    OperationsSessionTrain.train_id
                    == train_id,
                )
            )

            assignment = (
                session.execute(
                    statement
                )
                .scalars()
                .first()
            )

            if assignment is None:

                return (
                    False,
                    "Train is not assigned to this Operations Session.",
                )

            session.delete(
                assignment
            )

            session.commit()

            return (
                True,
                "Train removed from Operations Session.",
            )

    #
    # Delete all train assignments for
    # an Operations Session.
    #

    @staticmethod
    def delete_by_operations_session(
        operations_session_id,
    ):

        if operations_session_id is None:

            return (
                False,
                "No Operations Session was specified.",
            )

        with SessionLocal() as session:

            statement = (
                select(
                    OperationsSessionTrain
                )
                .where(
                    OperationsSessionTrain.operations_session_id
                    == operations_session_id
                )
            )

            assignments = (
                session.execute(
                    statement
                )
                .scalars()
                .all()
            )

            for assignment in assignments:

                session.delete(
                    assignment
                )

            session.commit()

            return (
                True,
                (
                    f"Deleted {len(assignments)} "
                    "train assignment(s)."
                ),
            )

    #
    # Delete all Operations Session assignments
    # for a Train.
    #

    @staticmethod
    def delete_by_train(
        train_id,
    ):

        if train_id is None:

            return (
                False,
                "No train was specified.",
            )

        with SessionLocal() as session:

            statement = (
                select(
                    OperationsSessionTrain
                )
                .where(
                    OperationsSessionTrain.train_id
                    == train_id
                )
            )

            assignments = (
                session.execute(
                    statement
                )
                .scalars()
                .all()
            )

            for assignment in assignments:

                session.delete(
                    assignment
                )

            session.commit()

            return (
                True,
                (

                    "train assignment(s)."

            )
